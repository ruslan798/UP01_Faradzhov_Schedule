    user_account {
        int id PK "IDENTITY"
        nvarchar login "UNIQUE"
        nvarchar password_hash "SHA-1"
        nvarchar role "Admin / User"
        nvarchar full_name "ФИО"
        nvarchar email "Email"
        nvarchar phone "Телефон"
        nvarchar photo "Путь к фото"
    }

    department {
        int id PK "IDENTITY"
        nvarchar name "Название отделения"
        nvarchar code "Код (ИСИП, КСК)"
        nvarchar head_name "Заведующий"
    }

    teacher {
        int id PK "IDENTITY"
        int user_id FK "Учетная запись"
        int department_id FK "Отделение"
        nvarchar full_name "ФИО преподавателя"
        nvarchar academic_degree "Степень / Должность"
        nvarchar email "Email"
        nvarchar phone "Телефон"
        nvarchar photo "Фото"
    }

    classroom {
        int id PK "IDENTITY"
        nvarchar room_number "Номер аудитории"
        nvarchar building "Корпус"
        int capacity "Вместимость"
        nvarchar room_type "Тип (компьютерный/лекция)"
        bit has_projector "Проектор"
    }

    study_group {
        int id PK "IDENTITY"
        nvarchar group_name "Название (4ИСИП-523)"
        int department_id FK "Отделение"
        int course_number "Курс (1-4)"
        int student_count "Кол-во студентов"
    }

    subject {
        int id PK "IDENTITY"
        nvarchar name "Дисциплина"
        int total_hours "Академические часы"
        int department_id FK "Отделение"
    }

    time_slot {
        int id PK "IDENTITY"
        int pair_number "Номер пары (1-6)"
        nvarchar start_time "Начало (08:30)"
        nvarchar end_time "Окончание (10:00)"
    }

    schedule_item {
        int id PK "IDENTITY"
        nvarchar day_of_week "День недели"
        int time_slot_id FK "Пара"
        int group_id FK "Группа"
        int subject_id FK "Дисциплина"
        int teacher_id FK "Преподаватель"
        int classroom_id FK "Аудитория"
        nvarchar lesson_type "Тип (Лекция/Лаб)"
        nvarchar week_type "Неделя (Все/Числитель)"
        nvarchar status "Статус (Утверждено)"
        datetime created_at "Дата создания"
    }
